        </select>
        <button onclick="adicionarAluno()">Cadastrar Aluno</button>

<script>
    let alunos = JSON.parse(localStorage.getItem('alunosAcademia')) || [];

    function renderizarTabela() {
        const tbody = document.getElementById('listaAlunos');
        tbody.innerHTML = '';

        alunos.forEach((aluno, index) => {
            const classeStatus = aluno.status === 'Ativo' ? 'status-ativo' : 'status-pendente';
            tbody.innerHTML += `
                <tr>
                    <td>${aluno.nome}</td>
                    <td>${aluno.plano}</td>
                    <td class="${classeStatus}">${aluno.status}</td>
                    <td><button class="btn-excluir" onclick="removerAluno(${index})">Excluir</button></td>
                </tr>
            `;
        });
    }

    function adicionarAluno() {
        const nome = document.getElementById('nome').value.trim();
        const plano = document.getElementById('plano').value;
        const status = document.getElementById('status').value;

        if (!nome) {
            alert('Por favor, digite o nome do aluno!');
            return;
        }

        alunos.push({ nome, plano, status });
        localStorage.setItem('alunosAcademia', JSON.stringify(alunos));
        document.getElementById('nome').value = '';
        renderizarTabela();
    }

    function removerAluno(index) {
        alunos.splice(index, 1);
        localStorage.setItem('alunosAcademia', JSON.stringify(alunos));
        renderizarTabela();
    }

    renderizarTabela();
</script>
